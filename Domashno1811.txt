            List<int> nums=Console.ReadLine()
                .Split(' ')
                .Select(int.Parse)
                .ToList();
            Console.WriteLine($"Първи списък: {string.Join(" ",nums)}");

            int numberToAdd = int.Parse(Console.ReadLine());
            int numberToRemove = int.Parse(Console.ReadLine());
            int indexToRemove = int.Parse(Console.ReadLine());

            nums.Add(numberToAdd);
            Console.WriteLine($"След Add: {string.Join(" ", nums)}");

            nums.Insert(0, numberToAdd);
            Console.WriteLine($"След Insert: {string.Join(" ",nums)}");

            if (nums.Contains(numberToRemove))
            {
                nums.Remove(numberToRemove);
            }
            else
            {
                Console.WriteLine($"Числото {numberToRemove} липсва в списъка.");
            }
            Console.WriteLine($"След Remove: {string.Join(" ", nums)}");

            if (indexToRemove >= 0 && indexToRemove < nums.Count)
            {
                nums.RemoveAt(indexToRemove);
            }
            else
            {
                Console.WriteLine("Невалиден Index");
            }
            Console.WriteLine($"След RemoveAt: {string.Join(" ", nums)}");

            Console.WriteLine($"Краен брой елементи: {nums.Count}");
            for(int i =0; i < nums.Count; i++)
            {
                Console.WriteLine($"{i}.{nums[i]}");
            }
